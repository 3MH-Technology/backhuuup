HF Space يحظر الاتصال بـ api.telegram.org. استخدم البروكسي المدمج:

import telebot
import os

bot = telebot.TeleBot("TOKEN")
# استخدم البروكسي المدمج:
proxy = os.environ.get("TG_PROXY", "")
if proxy:
  from telebot import apihelper
  apihelper.API_URL = proxy + "/{token}/{method}"

@bot.message_handler(commands=['start'])
def start(msg):
  bot.reply_to(msg, 'مرحباً!')

bot.polling()
